## AES256-CBC Payload Decryption via bCrypt

Decrypts an AES256-CBC encrypted payload buffer using the Windows bCrypt library. `InstallAesDecryption` drives the same CNG provider setup as the encryption counterpart, but substitutes `BCryptDecrypt` for `BCryptEncrypt`. `SimpleDecryption` is a convenience wrapper that initialises the structure and returns the recovered plaintext buffer and its size to the caller.

**Usage**

Call `SimpleDecryption` with the ciphertext buffer and its size, the original 32-byte key, the original 16-byte IV, and two `OUT` pointers that receive the plaintext buffer address and its size. The plaintext buffer is heap-allocated by the function; the caller is responsible for freeing it with `HeapFree`.

- `IN PVOID pCipherTextData`  Base address of the encrypted payload buffer to decrypt.
- `IN DWORD sCipherTextSize`  Size of `pCipherTextData` in bytes.
- `IN PBYTE pKey`  32-byte AES256 key; must match the key used during encryption.
- `IN PBYTE pIv`  16-byte IV; must match the IV used during encryption.
- `OUT PVOID* pPlainTextData`  Receives the base address of the heap-allocated plaintext buffer.
- `OUT DWORD* sPlainTextSize`  Receives the size of the plaintext buffer in bytes.

**Return Value**

`BOOL`  Returns `TRUE` if decryption succeeds and `pPlainTextData` / `sPlainTextSize` are populated, otherwise `FALSE`.

**External Files**

- `windows.h`  `PBYTE`, `PVOID`, `DWORD`, `BOOL`, `NTSTATUS`, `ULONG`, `PUCHAR`, `IN`, `OUT`, `HeapAlloc()`, `HeapFree()`, `GetProcessHeap()`
- `bcrypt.h`  `BCryptOpenAlgorithmProvider()`, `BCryptGetProperty()`, `BCryptSetProperty()`, `BCryptGenerateSymmetricKey()`, `BCryptDecrypt()`, `BCryptDestroyKey()`, `BCryptCloseAlgorithmProvider()`, `NT_SUCCESS()`
- `stdio.h`  `printf()`

### Code

```c
#include <windows.h>
#include <bcrypt.h>
#include <stdio.h>

#pragma comment(lib, "bcrypt.lib")

#define KEYSIZE   32    // 32-byte key for AES256.
#define IVSIZE    16    // 16-byte IV for CBC mode.

typedef struct _AES {

    PBYTE   pPlainText;     // Base address of the decrypted output buffer.
    DWORD   dwPlainSize;    // Size of the decrypted output in bytes.

    PBYTE   pCipherText;    // Base address of the encrypted input buffer.
    DWORD   dwCipherSize;   // Size of the encrypted input in bytes.

    PBYTE   pKey;           // 32-byte AES256 key.
    PBYTE   pIv;            // 16-byte initialisation vector.

} AES, *PAES;


BOOL InstallAesDecryption(PAES pAes)
{
    BOOL                bSTATE      = TRUE;
    BCRYPT_ALG_HANDLE   hAlgorithm  = NULL;
    BCRYPT_KEY_HANDLE   hKeyHandle  = NULL;

    ULONG   cbResult    = NULL;
    DWORD   dwBlockSize = NULL;
    DWORD   cbKeyObject = NULL;
    PBYTE   pbKeyObject = NULL;
    PBYTE   pbPlainText = NULL;
    DWORD   cbPlainText = NULL;
    NTSTATUS STATUS     = NULL;

    // Load the AES CNG algorithm provider.
    STATUS = BCryptOpenAlgorithmProvider(
        &hAlgorithm,            // Receives the algorithm provider handle.
        BCRYPT_AES_ALGORITHM,   // Selects AES as the algorithm.
        NULL,                   // Use the default provider.
        0                       // No flags.
    );
    if (!NT_SUCCESS(STATUS)) {
        printf("[!] BCryptOpenAlgorithmProvider Failed With Error: 0x%0.8X \n", STATUS);
        bSTATE = FALSE; goto _EndOfFunc;
    }

    // Retrieve the required size of the key object buffer used by BCryptGenerateSymmetricKey.
    STATUS = BCryptGetProperty(
        hAlgorithm,             // Algorithm provider handle.
        BCRYPT_OBJECT_LENGTH,   // Property: size of the key object.
        (PBYTE)&cbKeyObject,    // Output: size in bytes.
        sizeof(DWORD),          // Size of the output buffer.
        &cbResult,              // Bytes written.
        0                       // No flags.
    );
    if (!NT_SUCCESS(STATUS)) {
        printf("[!] BCryptGetProperty[1] Failed With Error: 0x%0.8X \n", STATUS);
        bSTATE = FALSE; goto _EndOfFunc;
    }

    // Retrieve the block size of AES; must be 16 bytes.
    STATUS = BCryptGetProperty(
        hAlgorithm,             // Algorithm provider handle.
        BCRYPT_BLOCK_LENGTH,    // Property: cipher block size.
        (PBYTE)&dwBlockSize,    // Output: block size in bytes.
        sizeof(DWORD),          // Size of the output buffer.
        &cbResult,              // Bytes written.
        0                       // No flags.
    );
    if (!NT_SUCCESS(STATUS)) {
        printf("[!] BCryptGetProperty[2] Failed With Error: 0x%0.8X \n", STATUS);
        bSTATE = FALSE; goto _EndOfFunc;
    }

    if (dwBlockSize != 16) {
        bSTATE = FALSE; goto _EndOfFunc;
    }

    pbKeyObject = (PBYTE)HeapAlloc(GetProcessHeap(), 0, cbKeyObject);
    if (pbKeyObject == NULL) {
        bSTATE = FALSE; goto _EndOfFunc;
    }

    // Set the chaining mode to CBC.
    STATUS = BCryptSetProperty(
        hAlgorithm,                     // Algorithm provider handle.
        BCRYPT_CHAINING_MODE,           // Property: block cipher mode.
        (PBYTE)BCRYPT_CHAIN_MODE_CBC,   // Value: cipher block chaining.
        sizeof(BCRYPT_CHAIN_MODE_CBC),  // Size of the value.
        0                               // No flags.
    );
    if (!NT_SUCCESS(STATUS)) {
        printf("[!] BCryptSetProperty Failed With Error: 0x%0.8X \n", STATUS);
        bSTATE = FALSE; goto _EndOfFunc;
    }

    // Build the key object from the caller-supplied AES key.
    STATUS = BCryptGenerateSymmetricKey(
        hAlgorithm,         // Algorithm provider handle.
        &hKeyHandle,        // Receives the key handle.
        pbKeyObject,        // Buffer for the key object.
        cbKeyObject,        // Size of the key object buffer.
        (PBYTE)pAes->pKey,  // Raw key material (32 bytes for AES256).
        KEYSIZE,            // Size of the key material.
        0                   // No flags.
    );
    if (!NT_SUCCESS(STATUS)) {
        printf("[!] BCryptGenerateSymmetricKey Failed With Error: 0x%0.8X \n", STATUS);
        bSTATE = FALSE; goto _EndOfFunc;
    }

    // First call: retrieve the required plaintext buffer size.
    STATUS = BCryptDecrypt(
        hKeyHandle,                     // Key handle.
        (PUCHAR)pAes->pCipherText,      // Input ciphertext buffer.
        (ULONG)pAes->dwCipherSize,      // Size of the ciphertext buffer.
        NULL,                           // No padding info (CBC does not use this).
        pAes->pIv,                      // 16-byte IV; consumed by the function.
        IVSIZE,                         // Size of the IV.
        NULL,                           // NULL output: size-query call.
        0,                              // Output buffer size (0 for query).
        &cbPlainText,                   // Receives the required output buffer size.
        BCRYPT_BLOCK_PADDING            // Strip PKCS padding from the final block.
    );
    if (!NT_SUCCESS(STATUS)) {
        printf("[!] BCryptDecrypt[1] Failed With Error: 0x%0.8X \n", STATUS);
        bSTATE = FALSE; goto _EndOfFunc;
    }

    pbPlainText = (PBYTE)HeapAlloc(GetProcessHeap(), 0, cbPlainText);
    if (pbPlainText == NULL) {
        bSTATE = FALSE; goto _EndOfFunc;
    }

    // Second call: decrypt the payload and store the result in pbPlainText.
    STATUS = BCryptDecrypt(
        hKeyHandle,                     // Key handle.
        (PUCHAR)pAes->pCipherText,      // Input ciphertext buffer.
        (ULONG)pAes->dwCipherSize,      // Size of the ciphertext buffer.
        NULL,                           // No padding info.
        pAes->pIv,                      // 16-byte IV.
        IVSIZE,                         // Size of the IV.
        pbPlainText,                    // Output buffer for the plaintext.
        cbPlainText,                    // Size of the output buffer.
        &cbResult,                      // Bytes written to pbPlainText.
        BCRYPT_BLOCK_PADDING            // Strip PKCS padding from the final block.
    );
    if (!NT_SUCCESS(STATUS)) {
        printf("[!] BCryptDecrypt[2] Failed With Error: 0x%0.8X \n", STATUS);
        bSTATE = FALSE; goto _EndOfFunc;
    }

_EndOfFunc:
    if (hKeyHandle)
        BCryptDestroyKey(hKeyHandle);
    if (hAlgorithm)
        BCryptCloseAlgorithmProvider(hAlgorithm, 0);
    if (pbKeyObject)
        HeapFree(GetProcessHeap(), 0, pbKeyObject);
    if (pbPlainText != NULL && bSTATE) {
        pAes->pPlainText  = pbPlainText;
        pAes->dwPlainSize = cbPlainText;
    }
    return bSTATE;
}


BOOL SimpleDecryption(IN PVOID pCipherTextData, IN DWORD sCipherTextSize, IN PBYTE pKey, IN PBYTE pIv, OUT PVOID* pPlainTextData, OUT DWORD* sPlainTextSize)
{
    if (pCipherTextData == NULL || sCipherTextSize == NULL || pKey == NULL || pIv == NULL)
        return FALSE;

    AES Aes = {
        .pKey         = pKey,
        .pIv          = pIv,
        .pCipherText  = pCipherTextData,
        .dwCipherSize = sCipherTextSize
    };

    if (!InstallAesDecryption(&Aes))
        return FALSE;

    *pPlainTextData = Aes.pPlainText;
    *sPlainTextSize = Aes.dwPlainSize;

    return TRUE;
}
```

**Example**

```c
unsigned char pKey[32]       = { 0 };  // Must match the encryption key.
unsigned char pIv[16]        = { 0 };  // Must match the encryption IV.
unsigned char pCipherText[] = { 0x12, 0x34, 0x56, 0x78, 0x9A, 0xBC };  // Output of the Encryption note.

PVOID pPlainTextData = NULL;
DWORD sPlainTextSize  = 0;

// Decrypt the ciphertext; pPlainTextData is heap-allocated by the function.
if (SimpleDecryption(pCipherText, sizeof(pCipherText), pKey, pIv,
                     &pPlainTextData, &sPlainTextSize)) {
    HeapFree(GetProcessHeap(), 0, pPlainTextData);  // Caller frees.
}
```
