## AES256-CBC Payload Encryption via Tiny-AES

Encrypts a plaintext payload using the [tiny-AES-c](https://github.com/kokke/tiny-AES-c) library without any WinAPI cryptographic calls. A custom `PaddBuffer` helper pads the input to the nearest multiple of 16 bytes when required, since Tiny-AES does not support automatic padding. `AES_init_ctx_iv` initialises the cipher context and `AES_CBC_encrypt_buffer` encrypts the buffer in place. The caller receives a pointer to the encrypted (and possibly padded) output buffer.

**Usage**

Call `TinyAesEncryptPayload` with a 32-byte key, a 16-byte IV, the plaintext buffer and its size, and two `OUT` pointers that receive the encrypted buffer address and its size. The output buffer is heap-allocated by the function; the caller is responsible for freeing it with `HeapFree`.

- `IN PBYTE pKey`  32-byte AES256 key. Requires `AES256` defined in `aes.h`.
- `IN PBYTE pIv`  16-byte initialisation vector for CBC mode.
- `IN PBYTE pPlainTextData`  Base address of the plaintext payload buffer.
- `IN SIZE_T sPlainTextSize`  Size of `pPlainTextData` in bytes.
- `OUT PBYTE* pCipherTextData`  Receives the base address of the heap-allocated encrypted output buffer.
- `OUT SIZE_T* sCipherTextSize`  Receives the size of the encrypted output buffer in bytes; may be larger than `sPlainTextSize` due to padding.

**Return Value**

`BOOL`  Returns `TRUE` if encryption succeeds and `pCipherTextData` / `sCipherTextSize` are populated, otherwise `FALSE`.

**External Files**

- [aes.h](<0.files/18.AES Encryption via Tiny-AES/aes.h>) / [aes.c](<0.files/18.AES Encryption via Tiny-AES/aes.c>)  ⚠️ Must be added manually , set `#define AES256 1` in `aes.h` and add `aes.c` to the project. Provides `AES_init_ctx_iv()`, `AES_CBC_encrypt_buffer()`.
- `windows.h`  `PBYTE`, `BOOL`, `SIZE_T`, `IN`, `OUT`, `HeapAlloc()`, `HeapFree()`, `GetProcessHeap()`, `ZeroMemory()`
- `string.h`  `memcpy()`

### Code

```c
#include <windows.h>
#include <string.h>
#include "aes.h"

// aes.h must have AES256 enabled:
//   //#define AES128 1
//   //#define AES192 1
//   #define AES256 1

#define KEYSIZE   32    // 32-byte key for AES256.
#define IVSIZE    16    // 16-byte IV for CBC mode.


BOOL PaddBuffer(IN PBYTE InputBuffer, IN SIZE_T InputBufferSize, OUT PBYTE* OutputPaddedBuffer, OUT SIZE_T* OutputPaddedSize)
{
    PBYTE  PaddedBuffer = NULL;
    SIZE_T PaddedSize   = NULL;

    // Round up to the nearest multiple of 16 bytes.
    PaddedSize   = InputBufferSize + 16 - (InputBufferSize % 16);

    PaddedBuffer = (PBYTE)HeapAlloc(GetProcessHeap(), 0, PaddedSize);
    if (!PaddedBuffer)
        return FALSE;

    // Zero the allocation so unused trailing bytes are clean.
    ZeroMemory(PaddedBuffer, PaddedSize);

    // Copy the original payload into the padded buffer.
    memcpy(PaddedBuffer, InputBuffer, InputBufferSize);

    *OutputPaddedBuffer = PaddedBuffer;
    *OutputPaddedSize   = PaddedSize;

    return TRUE;
}


BOOL TinyAesEncryptPayload(IN PBYTE pKey, IN PBYTE pIv, IN PBYTE pPlainTextData, IN SIZE_T sPlainTextSize, OUT PBYTE* pCipherTextData, OUT SIZE_T* sCipherTextSize)
{
    if (pKey == NULL || pIv == NULL || pPlainTextData == NULL || sPlainTextSize == 0)
        return FALSE;

    struct AES_ctx ctx;

    PBYTE  PaddedBuffer = NULL;
    SIZE_T PaddedSize   = NULL;

    // Initialize the Tiny-AES context with the key and IV.
    AES_init_ctx_iv(
        &ctx,   // AES context structure to initialise.
        pKey,   // 32-byte AES256 key.
        pIv     // 16-byte IV for CBC mode.
    );

    if (sPlainTextSize % 16 != 0) {

        // Payload is not a multiple of 16 bytes — pad it first.
        if (!PaddBuffer(pPlainTextData, sPlainTextSize, &PaddedBuffer, &PaddedSize))
            return FALSE;

        // Encrypt the padded buffer in place.
        AES_CBC_encrypt_buffer(
            &ctx,           // Initialised AES context.
            PaddedBuffer,   // Buffer to encrypt; modified in place.
            PaddedSize      // Must be a multiple of 16 bytes.
        );

        *pCipherTextData = PaddedBuffer;
        *sCipherTextSize = PaddedSize;

    } else {

        // Payload is already a multiple of 16 bytes — allocate a copy to encrypt.
        PaddedBuffer = (PBYTE)HeapAlloc(GetProcessHeap(), 0, sPlainTextSize);
        if (!PaddedBuffer)
            return FALSE;

        memcpy(PaddedBuffer, pPlainTextData, sPlainTextSize);

        // Encrypt the copy in place.
        AES_CBC_encrypt_buffer(
            &ctx,           // Initialised AES context.
            PaddedBuffer,   // Buffer to encrypt; modified in place.
            sPlainTextSize  // Size is already a multiple of 16 bytes.
        );

        *pCipherTextData = PaddedBuffer;
        *sCipherTextSize = sPlainTextSize;
    }

    return TRUE;
}
```

**Example**

```c
#include "aes.h"  // From the 18.AES Encryption via Tiny-AES/ asset folder.

unsigned char pKey[32]       = { 0 };  // 32-byte AES256 key (define AES256 in aes.h).
unsigned char pIv[16]        = { 0 };  // 16-byte CBC IV.
unsigned char pPlainText[]  = { 0xFC, 0x48, 0x83, 0xE4, 0xF0, 0xE5 };

PBYTE  pCipherTextData = NULL;
SIZE_T sCipherTextSize  = 0;

// Encrypt the plaintext; output is heap-allocated and padded to 16 bytes.
if (TinyAesEncryptPayload(pKey, pIv, pPlainText, sizeof(pPlainText),
                          &pCipherTextData, &sCipherTextSize)) {
    HeapFree(GetProcessHeap(), 0, pCipherTextData);  // Caller frees.
}
```
