## AES256-CBC Payload Encryption via bCrypt

Encrypts a plaintext payload buffer using AES256-CBC through the Windows bCrypt library. `InstallAesEncryption` drives the CNG provider, allocates an output heap buffer, and stores the ciphertext back into the `AES` structure. `SimpleEncryption` is a convenience wrapper that initialises the structure and returns the encrypted buffer and its size to the caller.

**Usage**

Call `SimpleEncryption` with the plaintext buffer and its size, a 32-byte key, a 16-byte IV, and two `OUT` pointers that receive the ciphertext buffer address and its size. The ciphertext buffer is heap-allocated by the function; the caller is responsible for freeing it with `HeapFree`.

- `IN PVOID pPlainTextData`  Base address of the plaintext payload buffer to encrypt.
- `IN DWORD sPlainTextSize`  Size of `pPlainTextData` in bytes.
- `IN PBYTE pKey`  32-byte AES256 key.
- `IN PBYTE pIv`  16-byte initialisation vector for CBC mode.
- `OUT PVOID* pCipherTextData`  Receives the base address of the heap-allocated ciphertext buffer.
- `OUT DWORD* sCipherTextSize`  Receives the size of the ciphertext buffer in bytes; may be larger than `sPlainTextSize` due to block padding.

**Return Value**

`BOOL`  Returns `TRUE` if encryption succeeds and `pCipherTextData` / `sCipherTextSize` are populated, otherwise `FALSE`.

**External Files**

- `windows.h`  `PBYTE`, `PVOID`, `DWORD`, `BOOL`, `NTSTATUS`, `ULONG`, `PUCHAR`, `IN`, `OUT`, `HeapAlloc()`, `HeapFree()`, `GetProcessHeap()`
- `bcrypt.h`  `BCryptOpenAlgorithmProvider()`, `BCryptGetProperty()`, `BCryptSetProperty()`, `BCryptGenerateSymmetricKey()`, `BCryptEncrypt()`, `BCryptDestroyKey()`, `BCryptCloseAlgorithmProvider()`, `NT_SUCCESS()`
- `stdio.h`  `printf()`

### Code

```c
#include <windows.h>
#include <bcrypt.h>
#include <stdio.h>

#pragma comment(lib, "bcrypt.lib")

#define KEYSIZE   32    // 32-byte key for AES256.
#define IVSIZE    16    // 16-byte IV for CBC mode.

typedef struct _AES {

    PBYTE   pPlainText;     // Base address of the plaintext data.
    DWORD   dwPlainSize;    // Size of the plaintext data in bytes.

    PBYTE   pCipherText;    // Base address of the encrypted output buffer.
    DWORD   dwCipherSize;   // Size of the encrypted output (may differ from dwPlainSize due to padding).

    PBYTE   pKey;           // 32-byte AES256 key.
    PBYTE   pIv;            // 16-byte initialisation vector.

} AES, *PAES;


BOOL InstallAesEncryption(PAES pAes)
{
    BOOL                bSTATE       = TRUE;
    BCRYPT_ALG_HANDLE   hAlgorithm   = NULL;
    BCRYPT_KEY_HANDLE   hKeyHandle   = NULL;

    ULONG   cbResult     = NULL;
    DWORD   dwBlockSize  = NULL;
    DWORD   cbKeyObject  = NULL;
    PBYTE   pbKeyObject  = NULL;
    PBYTE   pbCipherText = NULL;
    DWORD   cbCipherText = NULL;
    NTSTATUS STATUS      = NULL;

    // Load the AES CNG algorithm provider.
    STATUS = BCryptOpenAlgorithmProvider(
        &hAlgorithm,            // Receives the algorithm provider handle.
        BCRYPT_AES_ALGORITHM,   // Selects AES as the algorithm.
        NULL,                   // Use the default provider.
        0                       // No flags.
    );
    if (!NT_SUCCESS(STATUS)) {
        printf("[!] BCryptOpenAlgorithmProvider Failed With Error: 0x%0.8X \n", STATUS);
        bSTATE = FALSE; goto _EndOfFunc;
    }

    // Retrieve the required size of the key object buffer used by BCryptGenerateSymmetricKey.
    STATUS = BCryptGetProperty(
        hAlgorithm,             // Algorithm provider handle.
        BCRYPT_OBJECT_LENGTH,   // Property: size of the key object.
        (PBYTE)&cbKeyObject,    // Output: size in bytes.
        sizeof(DWORD),          // Size of the output buffer.
        &cbResult,              // Bytes written.
        0                       // No flags.
    );
    if (!NT_SUCCESS(STATUS)) {
        printf("[!] BCryptGetProperty[1] Failed With Error: 0x%0.8X \n", STATUS);
        bSTATE = FALSE; goto _EndOfFunc;
    }

    // Retrieve the block size of AES; must be 16 bytes.
    STATUS = BCryptGetProperty(
        hAlgorithm,             // Algorithm provider handle.
        BCRYPT_BLOCK_LENGTH,    // Property: cipher block size.
        (PBYTE)&dwBlockSize,    // Output: block size in bytes.
        sizeof(DWORD),          // Size of the output buffer.
        &cbResult,              // Bytes written.
        0                       // No flags.
    );
    if (!NT_SUCCESS(STATUS)) {
        printf("[!] BCryptGetProperty[2] Failed With Error: 0x%0.8X \n", STATUS);
        bSTATE = FALSE; goto _EndOfFunc;
    }

    if (dwBlockSize != 16) {
        bSTATE = FALSE; goto _EndOfFunc;
    }

    pbKeyObject = (PBYTE)HeapAlloc(GetProcessHeap(), 0, cbKeyObject);
    if (pbKeyObject == NULL) {
        bSTATE = FALSE; goto _EndOfFunc;
    }

    // Set the chaining mode to CBC.
    STATUS = BCryptSetProperty(
        hAlgorithm,                     // Algorithm provider handle.
        BCRYPT_CHAINING_MODE,           // Property: block cipher mode.
        (PBYTE)BCRYPT_CHAIN_MODE_CBC,   // Value: cipher block chaining.
        sizeof(BCRYPT_CHAIN_MODE_CBC),  // Size of the value.
        0                               // No flags.
    );
    if (!NT_SUCCESS(STATUS)) {
        printf("[!] BCryptSetProperty Failed With Error: 0x%0.8X \n", STATUS);
        bSTATE = FALSE; goto _EndOfFunc;
    }

    // Build the key object from the caller-supplied AES key.
    STATUS = BCryptGenerateSymmetricKey(
        hAlgorithm,     // Algorithm provider handle.
        &hKeyHandle,    // Receives the key handle.
        pbKeyObject,    // Buffer for the key object.
        cbKeyObject,    // Size of the key object buffer.
        (PBYTE)pAes->pKey, // Raw key material (32 bytes for AES256).
        KEYSIZE,        // Size of the key material.
        0               // No flags.
    );
    if (!NT_SUCCESS(STATUS)) {
        printf("[!] BCryptGenerateSymmetricKey Failed With Error: 0x%0.8X \n", STATUS);
        bSTATE = FALSE; goto _EndOfFunc;
    }

    // First call: retrieve the required ciphertext buffer size.
    STATUS = BCryptEncrypt(
        hKeyHandle,                     // Key handle.
        (PUCHAR)pAes->pPlainText,       // Input plaintext buffer.
        (ULONG)pAes->dwPlainSize,       // Size of the plaintext buffer.
        NULL,                           // No padding info (CBC does not use this).
        pAes->pIv,                      // 16-byte IV; consumed by the function.
        IVSIZE,                         // Size of the IV.
        NULL,                           // NULL output: size-query call.
        0,                              // Output buffer size (0 for query).
        &cbCipherText,                  // Receives the required output buffer size.
        BCRYPT_BLOCK_PADDING            // Apply PKCS padding to the final block.
    );
    if (!NT_SUCCESS(STATUS)) {
        printf("[!] BCryptEncrypt[1] Failed With Error: 0x%0.8X \n", STATUS);
        bSTATE = FALSE; goto _EndOfFunc;
    }

    pbCipherText = (PBYTE)HeapAlloc(GetProcessHeap(), 0, cbCipherText);
    if (pbCipherText == NULL) {
        bSTATE = FALSE; goto _EndOfFunc;
    }

    // Second call: encrypt the payload and store the result in pbCipherText.
    STATUS = BCryptEncrypt(
        hKeyHandle,                     // Key handle.
        (PUCHAR)pAes->pPlainText,       // Input plaintext buffer.
        (ULONG)pAes->dwPlainSize,       // Size of the plaintext buffer.
        NULL,                           // No padding info.
        pAes->pIv,                      // 16-byte IV.
        IVSIZE,                         // Size of the IV.
        pbCipherText,                   // Output buffer for the ciphertext.
        cbCipherText,                   // Size of the output buffer.
        &cbResult,                      // Bytes written to pbCipherText.
        BCRYPT_BLOCK_PADDING            // Apply PKCS padding to the final block.
    );
    if (!NT_SUCCESS(STATUS)) {
        printf("[!] BCryptEncrypt[2] Failed With Error: 0x%0.8X \n", STATUS);
        bSTATE = FALSE; goto _EndOfFunc;
    }

_EndOfFunc:
    if (hKeyHandle)
        BCryptDestroyKey(hKeyHandle);
    if (hAlgorithm)
        BCryptCloseAlgorithmProvider(hAlgorithm, 0);
    if (pbKeyObject)
        HeapFree(GetProcessHeap(), 0, pbKeyObject);
    if (pbCipherText != NULL && bSTATE) {
        pAes->pCipherText  = pbCipherText;
        pAes->dwCipherSize = cbCipherText;
    }
    return bSTATE;
}


BOOL SimpleEncryption(IN PVOID pPlainTextData, IN DWORD sPlainTextSize, IN PBYTE pKey, IN PBYTE pIv, OUT PVOID* pCipherTextData, OUT DWORD* sCipherTextSize)
{
    if (pPlainTextData == NULL || sPlainTextSize == NULL || pKey == NULL || pIv == NULL)
        return FALSE;

    AES Aes = {
        .pKey        = pKey,
        .pIv         = pIv,
        .pPlainText  = pPlainTextData,
        .dwPlainSize = sPlainTextSize
    };

    if (!InstallAesEncryption(&Aes))
        return FALSE;

    *pCipherTextData = Aes.pCipherText;
    *sCipherTextSize = Aes.dwCipherSize;

    return TRUE;
}
```

**Example**

```c
unsigned char pKey[32]      = { 0 };  // 32-byte AES256 key.
unsigned char pIv[16]       = { 0 };  // 16-byte CBC IV.
unsigned char pPlainText[]  = { 0xFC, 0x48, 0x83, 0xE4, 0xF0, 0xE5 };

PVOID pCipherTextData = NULL;
DWORD sCipherTextSize  = 0;

// Encrypt the plaintext; pCipherTextData is heap-allocated by the function.
if (SimpleEncryption(pPlainText, sizeof(pPlainText), pKey, pIv,
                     &pCipherTextData, &sCipherTextSize)) {
    HeapFree(GetProcessHeap(), 0, pCipherTextData);  // Caller frees.
}
```
