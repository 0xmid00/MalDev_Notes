## API Hashing - Custom GetModuleHandle & GetProcAddress

Replaces `GetModuleHandle` and `GetProcAddress` with hash-based equivalents that resolve DLLs and functions at runtime without leaving any name strings in the binary. `HashStringJenkinsOneAtATime32BitA/W` implement the Jenkins One-At-A-Time 32-bit algorithm; `HASHA`/`HASHW` macros wrap them for clean call sites. `GetModuleHandleH` walks `PEB->Ldr->InMemoryOrderModuleList`, uppercases each `FullDllName`, hashes it, and returns `DllBase` on a match — the input hash **must be of an uppercase DLL name**. `GetProcAddressH` walks the PE export directory of the supplied module base, hashes each exported function name with `HASHA`, and returns the function address on a match.

**Usage**

Obtain hash constants once by calling `HASHA` at runtime and printing the result, then hardcode them as `#define` values. Both functions must use the same `INITIAL_SEED` as the precomputed constants.

`GetModuleHandleH`
- `IN DWORD dwModuleNameHash`  Precomputed hash of the **uppercase** DLL base name (e.g. `HASHA("USER32.DLL")`).

`GetProcAddressH`
- `IN HMODULE hModule`  Base address of the DLL returned by `GetModuleHandleH`.
- `IN DWORD dwApiNameHash`  Precomputed hash of the function name (e.g. `HASHA("MessageBoxA")`).

**Return Value**

`GetModuleHandleH` → `HMODULE`  Base address of the matched DLL, or `NULL`.
`GetProcAddressH`  → `FARPROC`  Address of the matched function, or `NULL`.

**External Files**

- `windows.h`  `HMODULE`, `FARPROC`, `PPEB`, `PPEB_LDR_DATA`, `PLDR_DATA_TABLE_ENTRY`, `PBYTE`, `PDWORD`, `PWORD`, `DWORD`, `CHAR`, `PVOID`, `UINT32`, `SIZE_T`, `MAX_PATH`, `IN`, `PIMAGE_DOS_HEADER`, `PIMAGE_NT_HEADERS`, `PIMAGE_EXPORT_DIRECTORY`, `IMAGE_DOS_SIGNATURE`, `IMAGE_NT_SIGNATURE`, `IMAGE_DIRECTORY_ENTRY_EXPORT`, `__readgsqword()`, `__readfsdword()`, `toupper()`, `lstrlenA()`, `lstrlenW()`
- `winternl.h`  `PEB`, `PEB_LDR_DATA`, `LDR_DATA_TABLE_ENTRY`

### Code

```c
#include <windows.h>
#include <winternl.h>


// -----------------------------------------------------------------------
// Seed — must match the value used when precomputing hash constants.
// -----------------------------------------------------------------------
#define INITIAL_SEED    7


// -----------------------------------------------------------------------
// Jenkins One-At-A-Time 32-bit — ASCII and Unicode variants.
// -----------------------------------------------------------------------

UINT32 HashStringJenkinsOneAtATime32BitA(_In_ PCHAR String) {

    SIZE_T  Index   = 0;
    UINT32  Hash    = 0;
    SIZE_T  Length  = lstrlenA(String);

    while (Index != Length) {
        Hash += String[Index++];
        Hash += Hash << INITIAL_SEED;
        Hash ^= Hash >> 6;
    }

    Hash += Hash << 3;
    Hash ^= Hash >> 11;
    Hash += Hash << 15;

    return Hash;
}


UINT32 HashStringJenkinsOneAtATime32BitW(_In_ PWCHAR String) {

    SIZE_T  Index   = 0;
    UINT32  Hash    = 0;
    SIZE_T  Length  = lstrlenW(String);

    while (Index != Length) {
        Hash += String[Index++];
        Hash += Hash << INITIAL_SEED;
        Hash ^= Hash >> 6;
    }

    Hash += Hash << 3;
    Hash ^= Hash >> 11;
    Hash += Hash << 15;

    return Hash;
}

// Convenience macros — pass any string literal directly.
#define HASHA(API) (HashStringJenkinsOneAtATime32BitA((PCHAR)(API)))
#define HASHW(API) (HashStringJenkinsOneAtATime32BitW((PWCHAR)(API)))


// -----------------------------------------------------------------------
// GetModuleHandleH — PEB walk using hash instead of DLL name string.
// dwModuleNameHash must be the hash of the UPPERCASE DLL name.
// -----------------------------------------------------------------------

HMODULE GetModuleHandleH(IN DWORD dwModuleNameHash) {

    if (dwModuleNameHash == NULL)
        return NULL;

#ifdef _WIN64
    PPEB pPeb = (PPEB)(__readgsqword(0x60));
#elif _WIN32
    PPEB pPeb = (PPEB)(__readfsdword(0x30));
#endif

    PPEB_LDR_DATA           pLdr = (PPEB_LDR_DATA)(pPeb->Ldr);
    PLDR_DATA_TABLE_ENTRY   pDte = (PLDR_DATA_TABLE_ENTRY)(pLdr->InMemoryOrderModuleList.Flink);

    while (pDte) {

        if (pDte->FullDllName.Length != NULL && pDte->FullDllName.Length < MAX_PATH) {

            // Uppercase the full DLL name before hashing so the caller
            // can always pass HASHA("NTDLL.DLL") regardless of case on disk.
            CHAR  UpperCaseDllName  [MAX_PATH];
            DWORD i                 = 0;

            while (pDte->FullDllName.Buffer[i]) {
                UpperCaseDllName[i] = (CHAR)toupper(pDte->FullDllName.Buffer[i]);
                i++;
            }
            UpperCaseDllName[i] = '\0';

            // Hash the uppercased name and compare against the input hash.
            if (HASHA(UpperCaseDllName) == dwModuleNameHash)
                return (HMODULE)pDte->Reserved2[0];     // DllBase via winternl.h layout.
        }
        else {
            break;
        }

        // Advance to the next LDR_DATA_TABLE_ENTRY in the linked list.
        pDte = *(PLDR_DATA_TABLE_ENTRY*)(pDte);
    }

    return NULL;
}


// -----------------------------------------------------------------------
// GetProcAddressH — PE export walk using hash instead of function name string.
// -----------------------------------------------------------------------

FARPROC GetProcAddressH(IN HMODULE hModule, IN DWORD dwApiNameHash) {

    if (hModule == NULL || dwApiNameHash == NULL)
        return NULL;

    PBYTE pBase = (PBYTE)hModule;

    // Validate DOS header.
    PIMAGE_DOS_HEADER pImgDosHdr = (PIMAGE_DOS_HEADER)pBase;
    if (pImgDosHdr->e_magic != IMAGE_DOS_SIGNATURE)
        return NULL;

    // Validate NT headers.
    PIMAGE_NT_HEADERS pImgNtHdrs = (PIMAGE_NT_HEADERS)(pBase + pImgDosHdr->e_lfanew);
    if (pImgNtHdrs->Signature != IMAGE_NT_SIGNATURE)
        return NULL;

    // Locate the export directory via the optional header data directory.
    IMAGE_OPTIONAL_HEADER   ImgOptHdr       = pImgNtHdrs->OptionalHeader;
    PIMAGE_EXPORT_DIRECTORY pImgExportDir   = (PIMAGE_EXPORT_DIRECTORY)(pBase
                                                + ImgOptHdr.DataDirectory[IMAGE_DIRECTORY_ENTRY_EXPORT].VirtualAddress);

    // Resolve the three export arrays: names, addresses, and ordinals.
    PDWORD  FunctionNameArray       = (PDWORD)(pBase + pImgExportDir->AddressOfNames);
    PDWORD  FunctionAddressArray    = (PDWORD)(pBase + pImgExportDir->AddressOfFunctions);
    PWORD   FunctionOrdinalArray    = (PWORD) (pBase + pImgExportDir->AddressOfNameOrdinals);

    for (DWORD i = 0; i < pImgExportDir->NumberOfFunctions; i++) {

        CHAR*   pFunctionName       = (CHAR*)(pBase + FunctionNameArray[i]);
        PVOID   pFunctionAddress    = (PVOID)(pBase + FunctionAddressArray[FunctionOrdinalArray[i]]);

        // Hash the exported function name and compare against the input hash.
        if (dwApiNameHash == HASHA(pFunctionName))
            return (FARPROC)pFunctionAddress;
    }

    return NULL;
}
```

**Example**

```c
// --- Step 1: run once to obtain constants, then remove these prints ---
printf("[i] Hash Of \"%s\" : 0x%0.8X \n", "USER32.DLL",  HASHA("USER32.DLL"));
printf("[i] Hash Of \"%s\" : 0x%0.8X \n", "MessageBoxA", HASHA("MessageBoxA"));
// [i] Hash Of "USER32.DLL"  : 0x81E3778E
// [i] Hash Of "MessageBoxA" : 0xF10E27CA

// --- Step 2: hardcode the results ---
#define USER32DLL_HASH      0x81E3778E
#define MessageBoxA_HASH    0xF10E27CA

typedef int (WINAPI* fnMessageBoxA)(HWND, LPCSTR, LPCSTR, UINT);

// --- Step 3: resolve and call without any string in the binary ---
LoadLibraryA("USER32.DLL");     // Ensure the DLL is loaded before searching.

HMODULE       hUser32     = GetModuleHandleH(USER32DLL_HASH);
fnMessageBoxA pMessageBox = (fnMessageBoxA)GetProcAddressH(hUser32, MessageBoxA_HASH);

pMessageBox(NULL, "No strings exposed", "API Hashing", MB_OK);
```
