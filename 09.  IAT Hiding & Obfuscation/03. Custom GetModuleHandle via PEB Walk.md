## Custom GetModuleHandle via PEB Walk

Replaces `GetModuleHandle` by walking the PEB's `InMemoryOrderModuleList` entirely in userland without importing any WinAPI. Three interchangeable implementations are provided — uncomment the one to use. All three read the PEB address from the `GS` register (x64) or `FS` register (x86), walk to `Ldr->InMemoryOrderModuleList`, and iterate `LDR_DATA_TABLE_ENTRY` nodes comparing the DLL name case-insensitively until a match is found, then return `DllBase` as the `HMODULE`.

**Method differences:**

- **Method 1** — uses `winternl.h`'s limited struct + `offsetof` to correct the `Flink` offset + manual pointer arithmetic to reach `BaseDllName` (which Microsoft hides as `Reserved4/5`). No custom headers needed.
- **Method 2** — uses the full `LDR_DATA_TABLE_ENTRY` from `Structs.h` with `BaseDllName` named directly. Traverses via raw pointer dereference.
- **Method 3** — same full struct as Method 2 but traverses using explicit `pListHead`/`pListNode` doubly-linked-list logic. Cleaner termination condition.

**Usage**

Call any variant with the DLL base name in any case. Returns the `HMODULE` (base address) of the loaded DLL, or `NULL` if not found. Pass just the DLL name — no full path needed.

- `IN LPCWSTR szModuleName`  Base name of the DLL to find (e.g. `L"ntdll.dll"`). Case-insensitive.

**Return Value**

`HMODULE`  Base address of the matched loaded DLL, or `NULL` if not found.

**External Files**

- `windows.h`  `HMODULE`, `PPEB`, `PPEB_LDR_DATA`, `PLDR_DATA_TABLE_ENTRY`, `PUNICODE_STRING`, `PLIST_ENTRY`, `LPCWSTR`, `IN`, `WCHAR`, `MAX_PATH`, `__readgsqword()`, `__readfsdword()`, `lstrcmpiW()`, `lstrlenW()`
- `winternl.h`  Microsoft's `LDR_DATA_TABLE_ENTRY` (Method 1 only — do **not** include alongside `Structs.h`)
- [Structs.h](<0.files/CustomGetModuleHandle/Structs.h>)  ⚠️ Must be added manually : full `LDR_DATA_TABLE_ENTRY` with `BaseDllName`, `PEB_LDR_DATA`, and `PEB` definitions. Required for Method 2 and Method 3. Do **not** include `winternl.h` alongside it.
- `stdio.h`  `printf()`, `wprintf()`

code:
```c
// -----------------------------------------------------------------------
// Choose ONE inclusion:
//   winternl.h  → Method 1 only  (Microsoft's limited struct)
//   Structs.h   → Method 2 & 3   (full struct with BaseDllName named)
// Never include both — they redefine the same types.
// -----------------------------------------------------------------------
#include <windows.h>
#include <stdio.h>

#include "Structs.h"        // Method 2 & 3 — comment out for Method 1

#ifndef STRUCTS
#include <winternl.h>       // Method 1 — uncomment when NOT using Structs.h
#endif


// -----------------------------------------------------------------------
// Helper: case-insensitive wide-string comparison.
// Converts both strings to lowercase before comparing.
// -----------------------------------------------------------------------
BOOL IsStringEqual(IN LPCWSTR Str1, IN LPCWSTR Str2) {

    WCHAR   lStr1   [MAX_PATH],
            lStr2   [MAX_PATH];

    int     len1    = lstrlenW(Str1),
            len2    = lstrlenW(Str2);

    int     i = 0,
            j = 0;

    // Guard against buffer overflow.
    if (len1 >= MAX_PATH || len2 >= MAX_PATH)
        return FALSE;

    // Lowercase Str1.
    for (i = 0; i < len1; i++)
        lStr1[i] = (WCHAR)tolower(Str1[i]);
    lStr1[i++] = L'\0';

    // Lowercase Str2.
    for (j = 0; j < len2; j++)
        lStr2[j] = (WCHAR)tolower(Str2[j]);
    lStr2[j++] = L'\0';

    if (lstrcmpiW(lStr1, lStr2) == 0)
        return TRUE;

    return FALSE;
}


// -----------------------------------------------------------------------
// Method 1 — winternl.h + offsetof + manual BaseDllName pointer.
//
// Microsoft's LDR_DATA_TABLE_ENTRY hides BaseDllName inside Reserved4/5.
// offsetof corrects the Flink offset so pDteCurrent points to the real
// start of the struct. BaseDllName is then reached by stepping past
// FullDllName in memory using pointer arithmetic.
//
// Requires: winternl.h (NOT Structs.h)
// -----------------------------------------------------------------------
HMODULE GetModuleHandleReplacement(IN LPCWSTR szModuleName) {

#ifdef _WIN64
    PPEB pPeb = (PPEB)__readgsqword(0x60);
#elif _WIN32
    PPEB pPeb = (PPEB)__readfsdword(0x30);
#endif

    PPEB_LDR_DATA         pLdr        = (PPEB_LDR_DATA)pPeb->Ldr;

    // Correct the Flink offset: Flink points to InMemoryOrderLinks inside
    // the struct, not to the struct's start. offsetof adjusts for this.
    PLDR_DATA_TABLE_ENTRY pDteFirst   = (PLDR_DATA_TABLE_ENTRY)((PBYTE)pLdr->InMemoryOrderModuleList.Flink
                                            - offsetof(LDR_DATA_TABLE_ENTRY, InMemoryOrderLinks));
    PLDR_DATA_TABLE_ENTRY pDteCurrent = pDteFirst;

    do {
        if (pDteCurrent->FullDllName.Length != NULL) {

            // BaseDllName is not exposed by winternl.h — step past FullDllName
            // manually to reach it. Both are consecutive UNICODE_STRING fields.
            // FullDllName  = Path and name of module ,  BaseDllName  = Module name only.  
            PUNICODE_STRING pBaseDllName = (PUNICODE_STRING)((PBYTE)&pDteCurrent->FullDllName + sizeof(UNICODE_STRING));

            if (IsStringEqual(pBaseDllName->Buffer, szModuleName)) {
                wprintf(L"[+] Found \"%s\" at 0x%p \n", pBaseDllName->Buffer, pDteCurrent->DllBase);
                return (HMODULE)pDteCurrent->DllBase;
            }
        }

        // Advance to the next entry, correcting the offset each time.
        pDteCurrent = (PLDR_DATA_TABLE_ENTRY)((PBYTE)pDteCurrent->InMemoryOrderLinks.Flink
                        - offsetof(LDR_DATA_TABLE_ENTRY, InMemoryOrderLinks));

    } while (pDteCurrent != pDteFirst);

    return NULL;
}


// -----------------------------------------------------------------------
// Method 2 — full struct (Structs.h) + raw pointer traversal.
//
// Flink is cast directly to PLDR_DATA_TABLE_ENTRY without offsetof because
// MalDev's struct layout means the cast lands on InMemoryOrderLinks, making
// FullDllName resolve to BaseDllName. With Structs.h, BaseDllName is named
// directly and DllBase is at InInitializationOrderLinks.Flink.
//
// Requires: Structs.h (NOT winternl.h)
// -----------------------------------------------------------------------
HMODULE GetModuleHandleReplacement2(IN LPCWSTR szModuleName) {

#ifdef _WIN64
    PPEB pPeb = (PPEB)(__readgsqword(0x60));
#elif _WIN32
    PPEB pPeb = (PPEB)(__readfsdword(0x30));
#endif

    PPEB_LDR_DATA         pLdr = (PPEB_LDR_DATA)(pPeb->Ldr);
    PLDR_DATA_TABLE_ENTRY pDte = (PLDR_DATA_TABLE_ENTRY)(pLdr->InMemoryOrderModuleList.Flink);

    while (pDte) {

        if (pDte->FullDllName.Length != NULL) {

            if (IsStringEqual(pDte->FullDllName.Buffer, szModuleName)) {
                wprintf(L"[+] Found \"%s\" at 0x%p \n", pDte->FullDllName.Buffer, (HMODULE)(pDte->InInitializationOrderLinks.Flink));
                return (HMODULE)(pDte->InInitializationOrderLinks.Flink);
            }
        }
        else {
            break;
        }

        // Advance to the next node by dereferencing the current pointer.
        pDte = *(PLDR_DATA_TABLE_ENTRY*)(pDte);
    }

    return NULL;
}


// -----------------------------------------------------------------------
// Method 3 — full struct (Structs.h) + explicit doubly-linked-list traversal.
//
// Uses pListHead and pListNode to walk the list explicitly. Terminates
// cleanly when pListNode wraps back to pListHead (end of circular list).
// Functionally identical to Method 2 but follows doubly-linked-list idiom.
//
// Requires: Structs.h (NOT winternl.h)
// -----------------------------------------------------------------------
HMODULE GetModuleHandleReplacement3(IN LPCWSTR szModuleName) {

#ifdef _WIN64
    PPEB pPeb = (PPEB)(__readgsqword(0x60));
#elif _WIN32
    PPEB pPeb = (PPEB)(__readfsdword(0x30));
#endif

    PLDR_DATA_TABLE_ENTRY pDte      = (PLDR_DATA_TABLE_ENTRY)(pPeb->Ldr->InMemoryOrderModuleList.Flink);

    // Head of the circular doubly-linked list — used as the termination sentinel.
    PLIST_ENTRY pListHead           = (PLIST_ENTRY)&pPeb->Ldr->InMemoryOrderModuleList;
    // Current node — starts at the first real entry.
    PLIST_ENTRY pListNode           = (PLIST_ENTRY)pListHead->Flink;

    do {
        if (pDte->FullDllName.Length != NULL) {

            if (IsStringEqual(pDte->FullDllName.Buffer, szModuleName)) {
                wprintf(L"[+] Found \"%s\" at 0x%p \n", pDte->FullDllName.Buffer, (HMODULE)(pDte->InInitializationOrderLinks.Flink));
                return (HMODULE)(pDte->InInitializationOrderLinks.Flink);
            }

            // Advance pDte and pListNode to the next entry.
            pDte      = (PLDR_DATA_TABLE_ENTRY)(pListNode->Flink);
            pListNode = (PLIST_ENTRY)pListNode->Flink;
        }

    // Stop when the node wraps back to the head — end of the circular list.
    } while (pListNode != pListHead);

    return NULL;
}
```

**Example**

```c
// All three return the same address — pick one per project.

HMODULE h1 = GetModuleHandleReplacement(L"ntdll.dll");   // winternl.h + offsetof
HMODULE h2 = GetModuleHandleReplacement2(L"ntdll.dll");  // Structs.h + raw pointer
HMODULE h3 = GetModuleHandleReplacement3(L"ntdll.dll");  // Structs.h + list head/node

LoadLibraryA("ntdll.dll");     // Ensure the DLL is loaded before searching.

// Verify against the real API.
HMODULE hReal = GetModuleHandleW(L"ntdll.dll");
```
