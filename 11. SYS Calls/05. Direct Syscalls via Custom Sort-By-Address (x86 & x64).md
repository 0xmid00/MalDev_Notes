## Direct Syscalls via Custom Sort-By-Address (x86 & x64)

A self-contained SysWhispers2-style direct syscall implementation requiring no external tool. `GetSyscallList` walks the PEB to locate `ntdll.dll` by hash, parses its export directory, collects every `Zw`-prefixed function's hash and RVA into `SYSCALL_LIST`, then bubble-sorts the entries by RVA in ascending order — the sorted index of each entry **is** its SSN. `GetSysCallNumber` looks up the SSN at runtime by matching the caller-supplied hash against the sorted list. Two ASM stubs are provided — x64 uses `syscall`, x86 handles both native (`INT 2Eh`) and WoW64 (`fs:[0C0h]`) paths. Add a new `PROC`/`ENDP` block and a matching prototype for each additional NT function.

⚠️ Function hashes must use `Zw`-prefixed names (e.g. `"ZwProtectVirtualMemory"`) — the sort uses `Zw` exports.
⚠️ The `ntdll.dll` name hash must match the casing on the target system. Three common variants are hardcoded in `GetSyscallList` — add others if needed.
⚠️ x64: MASM required — Project → Build Customizations → enable MASM, set `.asm` Item Type to "Microsoft Macro Assembler", platform must be x64.
⚠️ x86:  set `.MODEL flat, c` and ensure the project targets x86. 

⚠️ MASM Encoding Issue: Open `syscalls.asm` → File → Save As → Save with Encoding → select `US-ASCII (Codepage 20127)` and keep line endings as `CRLF`.

⚠️ LNK2026 / LNK1281 (SAFESEH): Project → Properties → Linker → Advanced → Image Has Safe Exception Handlers → set to `No (/SAFESEH:NO)`.
         


**Usage**

Call `GetSysCallNumber` with the precomputed `Zw` function hash to retrieve the SSN, or call the NT stub directly — the stub resolves its own SSN internally. Obtain hashes by calling `HashStringJenkinsOneAtATime32BitA("ZwFuncName")` once and printing the result, then hardcode as a `#define`.

`GetSysCallNumber`
- `DWORD FuncHash`  Precomputed Jenkins hash of the `Zw`-prefixed function name.

**Return Value**

`GetSysCallNumber` → `DWORD`  SSN (sorted index) of the matched syscall, or `-1` if not found.
NT stubs → `NTSTATUS`  Standard NT status code returned in `EAX` after `syscall`/`INT 2Eh`.

**External Files**

- `windows.h`  `HANDLE`, `PVOID`, `PSIZE_T`, `ULONG`, `PULONG`, `NTSTATUS`, `PPEB`, `PPEB_LDR_DATA`, `PLIST_ENTRY`, `UNICODE_STRING`, `PIMAGE_DOS_HEADER`, `PIMAGE_NT_HEADERS`, `PIMAGE_EXPORT_DIRECTORY`, `IMAGE_DIRECTORY_ENTRY_EXPORT`, `__readgsqword()`, `__readfsdword()`
- `winternl.h`  `PEB`, `PEB_LDR_DATA`
- `Header.h`  NT prototypes, `LDR_DATA_TABLE_ENTRY_UNDOC`, `SYSCALL_ENTRY`, `SYSCALL_LIST` — defined below.
- `Source_x64.asm`  x64 MASM stub — `MyFuncMain` dispatcher + `NtProtectVirtualMemory PROC`.
- `Source_x86.asm`  x86 MASM stub — `MyFuncMain` dispatcher + `NtProtectVirtualMemory PROC`.

#ifndef HEADER_H
#define HEADER_H

#define MAX_ENTRIES     500
// Convert a relative virtual address to a virtual address.
#define RVA2VA(Type, DllBase, Rva) (Type)((PBYTE)(DllBase) + (Rva))


// -----------------------------------------------------------------------
// NT function prototypes — one per stub in the .asm file.
// __cdecl prevents the linker from decorating the name.
// -----------------------------------------------------------------------
NTSTATUS __cdecl NtProtectVirtualMemory(
    HANDLE  ProcessHandle,          // RCX / stack[0]
    PVOID*  BaseAddress,            // RDX / stack[1]
    PSIZE_T RegionSize,             // R8  / stack[2]
    ULONG   NewProtection,          // R9  / stack[3]
    PULONG  OldProtection           // stack[4]
);


// -----------------------------------------------------------------------
// Extended LDR entry that exposes BaseDllName (hidden in winternl.h).
// -----------------------------------------------------------------------
typedef struct _LDR_DATA_TABLE_ENTRY_UNDOC {
    PVOID           Reserved1[2];
    LIST_ENTRY      InMemoryOrderLinks;
    PVOID           Reserved2[2];
    PVOID           DllBase;
    PVOID           EntryPoint;
    ULONG           SizeOfImage;
    UNICODE_STRING  FullDllName;
    UNICODE_STRING  BaseDllName;    // Not in the official winternl.h definition.
} LDR_DATA_TABLE_ENTRY_UNDOC, *PLDR_DATA_TABLE_ENTRY_UNDOC;


// -----------------------------------------------------------------------
// Syscall list structures.
// -----------------------------------------------------------------------
typedef struct _SYSCALL_ENTRY {
    DWORD Hash;         // Jenkins hash of the Zw-prefixed function name.
    DWORD RvaAddress;   // RVA of the function — used for sort order = SSN.
} SYSCALL_ENTRY, *PSYSCALL_ENTRY;

typedef struct _SYSCALL_LIST {
    DWORD           Count;
    SYSCALL_ENTRY   Entries[MAX_ENTRIES];
} SYSCALL_LIST, *PSYSCALL_LIST;


UINT32  HashStringJenkinsOneAtATime32BitA(_In_ PCHAR String);
BOOL    GetSyscallList();

#endif

**Code.c**

```c
#include <windows.h>
#include <winternl.h>
#include <stdio.h>
#include "Header.h"


#define INITIAL_SEED    7   // Must match the seed used for all precomputed hashes.


// Global syscall list — populated once by GetSyscallList().
SYSCALL_LIST SyscallList = { 0 };


// -----------------------------------------------------------------------
// Jenkins One-At-A-Time — ASCII (function names).
// -----------------------------------------------------------------------
UINT32 HashStringJenkinsOneAtATime32BitA(_In_ PCHAR String) {

    SIZE_T  Index   = 0;
    UINT32  Hash    = 0;
    SIZE_T  Length  = lstrlenA(String);

    while (Index != Length) {
        Hash += String[Index++];
        Hash += Hash << INITIAL_SEED;
        Hash ^= Hash >> 6;
    }

    Hash += Hash << 3;
    Hash ^= Hash >> 11;
    Hash += Hash << 15;

    return Hash;
}


// -----------------------------------------------------------------------
// Jenkins One-At-A-Time — Unicode (DLL names from PEB).
// -----------------------------------------------------------------------
UINT32 HashStringJenkinsOneAtATime32BitW(_In_ PWCHAR String) {

    SIZE_T  Index   = 0;
    UINT32  Hash    = 0;
    SIZE_T  Length  = lstrlenW(String);

    while (Index != Length) {
        Hash += String[Index++];
        Hash += Hash << INITIAL_SEED;
        Hash ^= Hash >> 6;
    }

    Hash += Hash << 3;
    Hash ^= Hash >> 11;
    Hash += Hash << 15;

    return Hash;
}


// -----------------------------------------------------------------------
// GetSyscallList — walks the PEB to find ntdll.dll, parses its export
// directory, collects all Zw-prefixed function hashes and RVAs, then
// bubble-sorts by RVA ascending. The sorted index == SSN.
// -----------------------------------------------------------------------
BOOL GetSyscallList() {

    // Return immediately if already populated.
    if (SyscallList.Count) return TRUE;

    // 1 — Locate ntdll.dll base via the PEB.
#ifdef _WIN64
    PPEB pPeb = (PPEB)__readgsqword(0x60);
#else
    PPEB pPeb = (PPEB)__readfsdword(0x30);
#endif

    PPEB_LDR_DATA   pLdr    = pPeb->Ldr;
    PVOID           DllBase = NULL;

    PLIST_ENTRY pHead    = &pLdr->InMemoryOrderModuleList;
    PLIST_ENTRY pCurrent = pHead->Flink;

    while (pHead != pCurrent) {

        PLDR_DATA_TABLE_ENTRY_UNDOC pDte = (PLDR_DATA_TABLE_ENTRY_UNDOC)
            CONTAINING_RECORD(pCurrent, LDR_DATA_TABLE_ENTRY_UNDOC, InMemoryOrderLinks);

        UINT32 DllHash = HashStringJenkinsOneAtATime32BitW(pDte->BaseDllName.Buffer);

        // Match any common casing variant of ntdll.dll.
        if (DllHash == 0x82ACBBB8 ||    // "ntdll.dll"
            DllHash == 0x522BD567 ||    // "NTDLL.dll"
            DllHash == 0x4898F593) {    // "NTDLL.DLL"
            DllBase = pDte->DllBase;
            break;
        }

        pCurrent = pCurrent->Flink;
    }

    if (!DllBase) return FALSE;

    // 2 — Parse the export directory.
    PIMAGE_DOS_HEADER       pDosHdr     = (PIMAGE_DOS_HEADER)DllBase;
    PIMAGE_NT_HEADERS       pNtHdrs     = RVA2VA(PIMAGE_NT_HEADERS, DllBase, pDosHdr->e_lfanew);
    PIMAGE_EXPORT_DIRECTORY pExportDir  = RVA2VA(PIMAGE_EXPORT_DIRECTORY, DllBase,
                                            pNtHdrs->OptionalHeader.DataDirectory[IMAGE_DIRECTORY_ENTRY_EXPORT].VirtualAddress);

    DWORD   NumberOfNames       = pExportDir->NumberOfNames;
    PDWORD  RvaNamesArray       = RVA2VA(PDWORD, DllBase, pExportDir->AddressOfNames);
    PDWORD  RvaFunctionArray    = RVA2VA(PDWORD, DllBase, pExportDir->AddressOfFunctions);
    PWORD   OrdinalArray        = RVA2VA(PWORD,  DllBase, pExportDir->AddressOfNameOrdinals);

    PSYSCALL_ENTRY  Entries = SyscallList.Entries;
    DWORD           Count   = 0;

    // 3 — Collect all Zw-prefixed exports.
    for (DWORD i = 0; i < NumberOfNames && Count < MAX_ENTRIES; i++) {

        PCHAR FuncName = RVA2VA(PCHAR, DllBase, RvaNamesArray[i]);

        // x64 is little-endian: "Zw" in memory reads as 'wZ' when cast to USHORT.
        if (*(USHORT*)FuncName == 'wZ') {
            Entries[Count].Hash       = HashStringJenkinsOneAtATime32BitA(FuncName);
            Entries[Count].RvaAddress = RvaFunctionArray[OrdinalArray[i]];
            Count++;
        }
    }

    SyscallList.Count = Count;

    // 4 — Bubble sort by RVA ascending. Sorted index == SSN.
    for (DWORD i = 0; i < Count - 1; i++) {
        for (DWORD j = 0; j < Count - i - 1; j++) {
            if (Entries[j].RvaAddress > Entries[j + 1].RvaAddress) {
                SYSCALL_ENTRY Temp  = Entries[j];
                Entries[j]          = Entries[j + 1];
                Entries[j + 1]      = Temp;
            }
        }
    }

    return TRUE;
}


// -----------------------------------------------------------------------
// GetSysCallNumber — returns the SSN (sorted index) for the given hash.
// Called from the ASM stubs via EXTERN_C linkage.
// -----------------------------------------------------------------------
EXTERN_C DWORD GetSysCallNumber(DWORD FuncHash) {

    if (!GetSyscallList()) return (DWORD)-1;

    for (DWORD i = 0; i < SyscallList.Count; i++) {
        if (FuncHash == SyscallList.Entries[i].Hash)
            return i;   // Index in the sorted list == SSN.
    }

    return (DWORD)-1;
}
```

**Source_x64.asm**

```asm
.data
FuncHash DWORD 0        ; Global — holds the hash for the current call.

.code
EXTERN GetSysCallNumber: PROC


; -----------------------------------------------------------------------
; MyFuncMain — saves registers, resolves SSN via GetSysCallNumber,
; restores registers, then issues the syscall.
; SSN is already in EAX (return value of GetSysCallNumber) before syscall.
; -----------------------------------------------------------------------
MyFuncMain PROC

    pop rax                 ; Remove the return address pushed by CALL NtXxx.
    mov [rsp+ 8], rcx       ; Save RCX (arg1).
    mov [rsp+16], rdx       ; Save RDX (arg2).
    mov [rsp+24], r8        ; Save R8  (arg3).
    mov [rsp+32], r9        ; Save R9  (arg4).
    sub rsp, 28h
    mov ecx, FuncHash       ; Pass hash as argument to GetSysCallNumber.
    call GetSysCallNumber   ; Returns SSN in EAX.
    add rsp, 28h
    mov rcx, [rsp+ 8]       ; Restore RCX.
    mov rdx, [rsp+16]       ; Restore RDX.
    mov r8,  [rsp+24]       ; Restore R8.
    mov r9,  [rsp+32]       ; Restore R9.
    mov r10, rcx            ; Backup RCX into R10 — required by syscall ABI.
    syscall                 ; Issue syscall — NTSTATUS returned in EAX.
    ret

MyFuncMain ENDP


; -----------------------------------------------------------------------
; NT stub — load hash into FuncHash and jump into MyFuncMain.
; Add one PROC/ENDP block per NT function needed.
; -----------------------------------------------------------------------
NtProtectVirtualMemory PROC
    mov FuncHash, 4273C39Bh ; Jenkins hash of "ZwProtectVirtualMemory".
    call MyFuncMain
NtProtectVirtualMemory ENDP

END
```

**Source_x86.asm**

```asm
.686
.XMM
.MODEL flat, c

ASSUME fs:_DATA

.data

.code

EXTERN GetSysCallNumber: PROC


; -----------------------------------------------------------------------
; MyFuncMain — resolves SSN via GetSysCallNumber, then issues the syscall.
; Handles both native (INT 2Eh) and WoW64 (fs:[0C0h]) paths.
; -----------------------------------------------------------------------
MyFuncMain PROC

    pop  eax                        ; Remove return address pushed by CALL NtXxx.
    call GetSysCallNumber           ; Hash already on stack — SSN returned in EAX.
    add  esp, 4                     ; Clean up the hash argument from the stack.
    mov  ecx, fs:[0C0h]             ; Check for WoW64 transition pointer.
    test ecx, ecx
    jne  _wow64

    ; Native x86 path.
    lea  edx, [esp+4h]              ; EDX points to the argument list on the stack.
    INT  02Eh                       ; Issue native syscall.
    ret

_wow64:
    ; WoW64 path.
    xor  ecx, ecx
    lea  edx, [esp+4h]              ; EDX points to the argument list on the stack.
    call dword ptr fs:[0C0h]        ; Call WoW64 transition thunk.
    ret

MyFuncMain ENDP


; -----------------------------------------------------------------------
; NT stub — push hash onto stack and call MyFuncMain.
; Add one PROC/ENDP block per NT function needed.
; -----------------------------------------------------------------------
NtProtectVirtualMemory PROC
    push 4273C39Bh                  ; Jenkins hash of "ZwProtectVirtualMemory".
    call MyFuncMain
NtProtectVirtualMemory ENDP

END
```

**Example**

```c
// --- Obtain hashes once, then hardcode ---
// printf("ZwProtectVirtualMemory : 0x%08X\n", HashStringJenkinsOneAtATime32BitA("ZwProtectVirtualMemory"));
// printf("ntdll.dll              : 0x%08X\n", HashStringJenkinsOneAtATime32BitW(L"ntdll.dll"));
// Output:
// ZwProtectVirtualMemory : 0x4273C39B
// ntdll.dll              : 0x82ACBBB8

#define ZwProtectVirtualMemory_HASH     0x4273C39Bu

// --- Verify SSN resolution ---
DWORD ssn = GetSysCallNumber(ZwProtectVirtualMemory_HASH);
printf("[+] SSN for NtProtectVirtualMemory = %lu\n", ssn);

// --- Call the NT stub directly ---
DWORD   dwOldProtect    = 0;
SIZE_T  sRegionSize     = 1024;
PBYTE   pBase           = VirtualAlloc(NULL, sRegionSize, MEM_COMMIT | MEM_RESERVE, PAGE_READONLY);

NTSTATUS status = NtProtectVirtualMemory(
    GetCurrentProcess(),
    (PVOID*)&pBase,
    &sRegionSize,
    PAGE_READWRITE,
    &dwOldProtect
);

printf("[+] NTSTATUS : 0x%08X\n", status);
printf("[+] Old protection : 0x%08X\n", dwOldProtect);

VirtualFree(pBase, 0, MEM_RELEASE);
```
