## API Hooking via Custom Code

Installs and removes inline API hooks without any third-party library by writing a small `mov + jmp` trampoline shellcode over the first bytes of the target function's prologue. A `HookSt` structure holds the target address, the detour address, the saved original bytes, and the old memory protection — everything needed to both install and cleanly restore the hook. `InitializeHookStruct` saves the original bytes and flips the prologue to `RWX`. `InstallHook` patches the address into the trampoline and writes it over the prologue. `RemoveHook` restores the original bytes and reverts the protection.

> Because the trampoline overwrites the prologue entirely there is no clean way to call the original function from inside the detour using the same pointer. Use the wide-char equivalent (e.g. `MessageBoxW` instead of `MessageBoxA`) or a different unhooking API as an alternative to avoid an infinite loop.

**Usage**

1. Declare a `HookSt` struct zeroed out and call `InitializeHookStruct` with the target function address, the detour function address, and a pointer to the struct.
2. Call `InstallHook` to write the trampoline — any call to the target function now executes the detour.
3. Call `RemoveHook` when done — the original bytes are restored.

- `IN PVOID pFunctionToHook`  Address of the target function whose prologue will be overwritten.
- `IN PVOID pFunctionToRun`  Address of the detour function to execute instead.
- `OUT PHookSt Hook`  Pointer to a caller-allocated `HookSt` structure that receives all hook state.

**Return Value**

`InitializeHookStruct` / `InstallHook` / `RemoveHook` → `BOOL`  Returns `TRUE` on success, otherwise `FALSE`.

**External Files**

- `windows.h`  `PVOID`, `PBYTE`, `BOOL`, `DWORD`, `BYTE`, `HWND`, `LPCSTR`, `UINT`, `INT`, `IN`, `OUT`, `WINAPI`, `VirtualProtect()`, `GetLastError()`
- `string.h`  `memcpy()`, `memset()`
- `stdio.h`  `printf()`

### Code

```c
#include <windows.h>
#include <string.h>
#include <stdio.h>

// Trampoline shellcode sizes:
//   x64: 10 bytes (mov r10) + 3 bytes (jmp r10)  = 13 bytes
//   x86:  5 bytes (mov eax) + 2 bytes (jmp eax)  =  7 bytes
#ifdef _M_X64
#define TRAMPOLINE_SIZE     13
#endif
#ifdef _M_IX86
#define TRAMPOLINE_SIZE     7
#endif

typedef unsigned long long  uint64_t;
typedef unsigned int        uint32_t;
typedef unsigned char       uint8_t;


// ── HookSt Structure ──────────────────────────────────────────────────────────
// Holds all state needed to install and cleanly remove one hook.
// Populate with InitializeHookStruct before calling InstallHook or RemoveHook.
typedef struct _HookSt {

    PVOID   pFunctionToHook;                // VA of the function whose prologue will be patched.
    PVOID   pFunctionToRun;                 // VA of the detour function to jump to.
    BYTE    pOriginalBytes[TRAMPOLINE_SIZE]; // Saved original prologue bytes — used by RemoveHook.
    DWORD   dwOldProtection;               // Saved page protection — restored by RemoveHook.

} HookSt, *PHookSt;


// ── InitializeHookStruct ──────────────────────────────────────────────────────
// Populates Hook with the target and detour addresses, saves the original
// prologue bytes, and flips the target's page protection to RWX so that
// InstallHook can write the trampoline without an access violation.
// Must be called once before InstallHook.
BOOL InitializeHookStruct(IN PVOID pFunctionToHook, IN PVOID pFunctionToRun, OUT PHookSt Hook)
{
    if (pFunctionToHook == NULL || pFunctionToRun == NULL || Hook == NULL)
        return FALSE;

    Hook->pFunctionToHook = pFunctionToHook;
    Hook->pFunctionToRun  = pFunctionToRun;

    // Save the original prologue bytes so RemoveHook can restore them exactly.
    memcpy(
        Hook->pOriginalBytes,   // Destination: saved-bytes buffer in the struct.
        pFunctionToHook,        // Source: first TRAMPOLINE_SIZE bytes of the target function.
        TRAMPOLINE_SIZE
    );

    // Flip the target function's page to RWX so the prologue can be overwritten.
    // The old protection is saved for restoration by RemoveHook.
    if (!VirtualProtect(
        pFunctionToHook,            // Base address of the region to change.
        TRAMPOLINE_SIZE,            // Number of bytes to change protection for.
        PAGE_EXECUTE_READWRITE,     // New protection: read, write, and execute.
        &Hook->dwOldProtection      // Receives the original page protection.
    )) {
        printf("[!] VirtualProtect Failed With Error : %d \n", GetLastError());
        return FALSE;
    }

    return TRUE;
}


// ── InstallHook ───────────────────────────────────────────────────────────────
// Builds the arch-specific trampoline shellcode, patches in the detour address,
// and writes it over the target function's prologue.
// InitializeHookStruct must be called first.
BOOL InstallHook(IN PHookSt Hook)
{
    if (Hook == NULL || Hook->pFunctionToHook == NULL || Hook->pFunctionToRun == NULL)
        return FALSE;

#ifdef _M_X64
    // 64-bit trampoline: mov r10, <addr> ; jmp r10
    // pFunctionToRun address is patched at offset 2 (after the 0x49 0xBA prefix).
    uint8_t uTrampoline[] = {
        0x49, 0xBA,                                     // mov r10, imm64  (prefix + opcode)
        0x00, 0x00, 0x00, 0x00, 0x00, 0x00, 0x00, 0x00,// placeholder: 8-byte detour address
        0x41, 0xFF, 0xE2                                // jmp r10
    };

    uint64_t uPatch = (uint64_t)(Hook->pFunctionToRun);
    memcpy(&uTrampoline[2], &uPatch, sizeof(uPatch));   // Write detour VA at offset 2.
#endif

#ifdef _M_IX86
    // 32-bit trampoline: mov eax, <addr> ; jmp eax
    // pFunctionToRun address is patched at offset 1 (after the 0xB8 opcode).
    uint8_t uTrampoline[] = {
        0xB8,                           // mov eax, imm32  (opcode)
        0x00, 0x00, 0x00, 0x00,         // placeholder: 4-byte detour address
        0xFF, 0xE0                      // jmp eax
    };

    uint32_t uPatch = (uint32_t)(Hook->pFunctionToRun);
    memcpy(&uTrampoline[1], &uPatch, sizeof(uPatch));   // Write detour VA at offset 1.
#endif

    // Overwrite the target function's prologue with the trampoline shellcode.
    memcpy(
        Hook->pFunctionToHook,  // Destination: target function's prologue (already RWX).
        uTrampoline,            // Source: trampoline shellcode with patched address.
        sizeof(uTrampoline)     // TRAMPOLINE_SIZE bytes.
    );

    return TRUE;
}


// ── RemoveHook ────────────────────────────────────────────────────────────────
// Restores the original prologue bytes saved by InitializeHookStruct,
// clears the saved-bytes buffer, and reverts the page protection.
BOOL RemoveHook(IN PHookSt Hook)
{
    if (Hook == NULL || Hook->pFunctionToHook == NULL || Hook->pOriginalBytes == NULL)
        return FALSE;

    DWORD dwOldProtection = NULL;

    // Restore the original prologue bytes over the trampoline shellcode.
    memcpy(
        Hook->pFunctionToHook,  // Destination: target function's (still RWX) prologue.
        Hook->pOriginalBytes,   // Source: saved original bytes from InitializeHookStruct.
        TRAMPOLINE_SIZE
    );

    // Zero the saved-bytes buffer so it cannot be re-used accidentally.
    memset(Hook->pOriginalBytes, '\0', TRAMPOLINE_SIZE);

    // Revert the page protection to what it was before InitializeHookStruct.
    if (!VirtualProtect(
        Hook->pFunctionToHook,  // Base address of the restored prologue region.
        TRAMPOLINE_SIZE,        // Number of bytes to change protection for.
        Hook->dwOldProtection,  // Original protection saved by InitializeHookStruct.
        &dwOldProtection        // Receives the current (RWX) protection — discarded.
    )) {
        printf("[!] VirtualProtect Failed With Error : %d \n", GetLastError());
        return FALSE;
    }

    // Clear the struct so it cannot be used after removal.
    Hook->pFunctionToHook  = NULL;
    Hook->pFunctionToRun   = NULL;
    Hook->dwOldProtection  = NULL;

    return TRUE;
}
```

**Example**

```c
// ── Detour function — runs instead of MessageBoxA when hooked ─────────────────
// Because the prologue is overwritten there is no trampoline pointer to call
// the original. Use a different API (MessageBoxW) to avoid an infinite loop.
INT WINAPI MyMessageBoxA(HWND hWnd, LPCSTR lpText, LPCSTR lpCaption, UINT uType)
{
    printf("[+] Original Parameters : \n");
    printf("\t - lpText    : %s\n", lpText);
    printf("\t - lpCaption : %s\n", lpCaption);

    // Call the wide-char variant — NOT MessageBoxA (would re-enter the hook).
    return MessageBoxW(hWnd, L"Malware Development Is Cool", L"Hooked MsgBox", uType);
}


// ── Main usage flow ───────────────────────────────────────────────────────────

HookSt st = { 0 };

// Step 1 — initialise: save bytes, flip to RWX.
if (!InitializeHookStruct(&MessageBoxA, &MyMessageBoxA, &st))
    return -1;

// Before hook — real MessageBoxA runs.
MessageBoxA(NULL, "What Do You Think About me?", "Original MsgBox", MB_OK | MB_ICONQUESTION);

// Step 2 — install: write trampoline over prologue.
printf("[i] Installing The Hook ... ");
if (!InstallHook(&st))
    return -1;
printf("[+] DONE \n");

// Hooked — MyMessageBoxA runs instead.
MessageBoxA(NULL, " i am  Bad", "Original MsgBox", MB_OK | MB_ICONWARNING);

// Step 3 — remove: restore bytes, revert protection.
printf("[i] Removing The Hook ... ");
if (!RemoveHook(&st))
    return -1;
printf("[+] DONE \n");

// After unhook — real MessageBoxA runs again.
MessageBoxA(NULL, "Normal MsgBox Again", "Original MsgBox", MB_OK | MB_ICONINFORMATION);
```

**Parameter Sources**

No external parameter sources — the target and detour function addresses are passed directly by the caller at runtime.
