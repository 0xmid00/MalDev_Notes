## Shellcode IPv6Fuscation Obfuscation

Converts a shellcode buffer into a C-style array of IPv6 address strings by grouping every 16 bytes into one IPv6 address, with bytes represented as hex segments. The obfuscated output is always printed to the console. Optionally, the generated string array and its element count are saved into caller-supplied `OUT` pointers if they are non-`NULL`. The shellcode buffer must be a multiple of 16 bytes.

**Usage**

Call `GenerateIpv6Output` with the shellcode buffer and its size. Pass valid pointers to `pIpv6Array` and `pNmbrOfElements` to receive the heap-allocated string array and its count, or pass `NULL` for both to use print-only mode. When the array is returned, the caller is responsible for freeing each string with `HeapFree` then freeing the array itself.

- `IN PBYTE pShellcode`  Base address of the shellcode buffer to obfuscate.
- `IN SIZE_T ShellcodeSize`  Size of `pShellcode` in bytes; must be a multiple of 16.
- `OUT CHAR*** pIpv6Array`  Optional. Receives the base address of the heap-allocated array of IPv6 string pointers. Pass `NULL` to skip.
- `OUT SIZE_T* pNmbrOfElements`  Optional. Receives the number of IPv6 strings in the array (`ShellcodeSize / 16`). Pass `NULL` to skip.

**Return Value**

`BOOL`  Returns `TRUE` if the output was generated successfully, otherwise `FALSE`.

**External Files**

- `windows.h`  `PBYTE`, `BOOL`, `SIZE_T`, `IN`, `OUT`, `HeapAlloc()`, `HeapFree()`, `GetProcessHeap()`
- `stdio.h`  `printf()`, `sprintf()`

### Code

```c
#include <windows.h>
#include <stdio.h>


// Helper: formats 16 raw bytes into a heap-allocated IPv6 address string.
// Caller is responsible for freeing the returned buffer with HeapFree.
char* GenerateIpv6(int a, int b, int c, int d, int e, int f, int g, int h,
                   int i, int j, int k, int l, int m, int n, int o, int p)
{
    // Each of the 4 segment pairs needs 32 bytes; full result needs 128 bytes.
    char Output0[32], Output1[32], Output2[32], Output3[32];

    // Allocate the output string on the heap so it outlives this stack frame.
    char* result = (char*)HeapAlloc(GetProcessHeap(), 0, 128);
    if (result == NULL)
        return NULL;

    // Build each 4-byte hex segment pair.
    sprintf(Output0, "%0.2X%0.2X:%0.2X%0.2X", a, b, c, d);   // Bytes  1–4.
    sprintf(Output1, "%0.2X%0.2X:%0.2X%0.2X", e, f, g, h);   // Bytes  5–8.
    sprintf(Output2, "%0.2X%0.2X:%0.2X%0.2X", i, j, k, l);   // Bytes  9–12.
    sprintf(Output3, "%0.2X%0.2X:%0.2X%0.2X", m, n, o, p);   // Bytes 13–16.

    // Combine the 4 segment pairs into a full IPv6 address string.
    sprintf(result, "%s:%s:%s:%s", Output0, Output1, Output2, Output3);

    return result;
}


BOOL GenerateIpv6Output(IN PBYTE pShellcode, IN SIZE_T ShellcodeSize, OUT CHAR*** pIpv6Array, OUT SIZE_T* pNmbrOfElements)
{
    if (pShellcode == NULL || ShellcodeSize == NULL || ShellcodeSize % 16 != 0)
        return FALSE;

    SIZE_T ElementCount = ShellcodeSize / 16;

    // Only allocate the output array if the caller supplied a non-NULL pointer.
    CHAR** Ipv6Array = NULL;
    if (pIpv6Array != NULL) {
        Ipv6Array = (CHAR**)HeapAlloc(GetProcessHeap(), 0, ElementCount * sizeof(CHAR*));
        if (Ipv6Array == NULL)
            return FALSE;
    }

    printf("char* Ipv6Array[%d] = { \n\t", (int)ElementCount);

    // 'c' tracks how many bytes have been consumed toward the current IPv6 address.
    // Starts at 16 to trigger generation on the very first iteration.
    int    c = 16, counter = 0;
    SIZE_T Index = 0;
    char*  IP = NULL;

    for (int i = 0; i < ShellcodeSize; i++) {

        // Every 16 bytes consumed, generate one IPv6 address.
        if (c == 16) {
            counter++;

            // Consume bytes [i .. i+15] to form one IPv6 address.
            IP = GenerateIpv6(
                pShellcode[i],      pShellcode[i + 1],  pShellcode[i + 2],  pShellcode[i + 3],
                pShellcode[i + 4],  pShellcode[i + 5],  pShellcode[i + 6],  pShellcode[i + 7],
                pShellcode[i + 8],  pShellcode[i + 9],  pShellcode[i + 10], pShellcode[i + 11],
                pShellcode[i + 12], pShellcode[i + 13], pShellcode[i + 14], pShellcode[i + 15]
            );
            if (IP == NULL) {
                // Free already allocated strings before returning.
                if (Ipv6Array != NULL) {
                    for (SIZE_T j = 0; j < Index; j++)
                        HeapFree(GetProcessHeap(), 0, Ipv6Array[j]);
                    HeapFree(GetProcessHeap(), 0, Ipv6Array);
                }
                return FALSE;
            }

            // Save the string pointer only if the caller requested the array.
            if (Ipv6Array != NULL)
                Ipv6Array[Index++] = IP;

            if (i == ShellcodeSize - 16) {
                // Last address — no trailing comma.
                printf("\"%s\"", IP);

                // If the array is not being saved, free the string now.
                if (Ipv6Array == NULL)
                    HeapFree(GetProcessHeap(), 0, IP);

                break;
            } else {
                printf("\"%s\", ", IP);

                // If the array is not being saved, free the string now.
                if (Ipv6Array == NULL)
                    HeapFree(GetProcessHeap(), 0, IP);
            }

            c = 1;

            // Insert a newline every 3 addresses to keep output readable.
            if (counter % 3 == 0)
                printf("\n\t");

        } else {
            c++;
        }
    }

    printf("\n};\n\n");

    // Save outputs only if the caller supplied non-NULL pointers.
    if (pIpv6Array != NULL)
        *pIpv6Array = Ipv6Array;

    if (pNmbrOfElements != NULL)
        *pNmbrOfElements = ElementCount;

    return TRUE;
}
```
