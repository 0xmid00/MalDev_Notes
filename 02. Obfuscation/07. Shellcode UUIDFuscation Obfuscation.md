## Shellcode UUIDFuscation Obfuscation

Converts a shellcode buffer into a C-style array of UUID strings by grouping every 16 bytes into one UUID. The first three UUID segments use little-endian byte ordering; the last two use big-endian. The obfuscated output is always printed to the console. Optionally, the generated string array and its element count are saved into caller-supplied `OUT` pointers if they are non-`NULL`. The shellcode buffer must be a multiple of 16 bytes.

**Usage**

Call `GenerateUuidOutput` with the shellcode buffer and its size. Pass valid pointers to `pUuidArray` and `pNmbrOfElements` to receive the heap-allocated string array and its count, or pass `NULL` for both to use print-only mode. When the array is returned, the caller is responsible for freeing each string with `HeapFree` then freeing the array itself.

- `IN PBYTE pShellcode`  Base address of the shellcode buffer to obfuscate.
- `IN SIZE_T ShellcodeSize`  Size of `pShellcode` in bytes; must be a multiple of 16.
- `OUT CHAR*** pUuidArray`  Optional. Receives the base address of the heap-allocated array of UUID string pointers. Pass `NULL` to skip.
- `OUT SIZE_T* pNmbrOfElements`  Optional. Receives the number of UUID strings in the array (`ShellcodeSize / 16`). Pass `NULL` to skip.

**Return Value**

`BOOL`  Returns `TRUE` if the output was generated successfully, otherwise `FALSE`.

**External Files**

- `windows.h`  `PBYTE`, `BOOL`, `SIZE_T`, `IN`, `OUT`, `HeapAlloc()`, `HeapFree()`, `GetProcessHeap()`
- `stdio.h`  `printf()`, `sprintf()`

### Code

```c
#include <windows.h>
#include <stdio.h>


// Helper: formats 16 raw bytes into a heap-allocated UUID string.
// Segments 1-3 are little-endian; segments 4-5 are big-endian.
// Caller is responsible for freeing the returned buffer with HeapFree.
char* GenerateUUid(int a, int b, int c, int d, int e, int f, int g, int h,
                   int i, int j, int k, int l, int m, int n, int o, int p)
{
    // Each UUID segment buffer is 32 bytes; full result needs 128 bytes.
    char Output0[32], Output1[32], Output2[32], Output3[32];

    // Allocate the output string on the heap so it outlives this stack frame.
    char* result = (char*)HeapAlloc(GetProcessHeap(), 0, 128);
    if (result == NULL)
        return NULL;

    // Segment 1: bytes 1-4 in little-endian order (d,c,b,a).
    sprintf(Output0, "%0.2X%0.2X%0.2X%0.2X", d, c, b, a);

    // Segment 2: bytes 5-6 little-endian (f,e), segment 3: bytes 7-8 little-endian (h,g).
    sprintf(Output1, "%0.2X%0.2X-%0.2X%0.2X", f, e, h, g);

    // Segment 4: bytes 9-12 big-endian.
    sprintf(Output2, "%0.2X%0.2X-%0.2X%0.2X", i, j, k, l);

    // Segment 5: bytes 13-16 big-endian.
    sprintf(Output3, "%0.2X%0.2X%0.2X%0.2X", m, n, o, p);

    // Combine all segments into the final UUID string.
    sprintf(result, "%s-%s-%s%s", Output0, Output1, Output2, Output3);

    return result;
}


BOOL GenerateUuidOutput(IN PBYTE pShellcode, IN SIZE_T ShellcodeSize, OUT CHAR*** pUuidArray, OUT SIZE_T* pNmbrOfElements)
{
    if (pShellcode == NULL || ShellcodeSize == NULL || ShellcodeSize % 16 != 0)
        return FALSE;

    SIZE_T ElementCount = ShellcodeSize / 16;

    // Only allocate the output array if the caller supplied a non-NULL pointer.
    CHAR** UuidArray = NULL;
    if (pUuidArray != NULL) {
        UuidArray = (CHAR**)HeapAlloc(GetProcessHeap(), 0, ElementCount * sizeof(CHAR*));
        if (UuidArray == NULL)
            return FALSE;
    }

    printf("char* UuidArray[%d] = { \n\t", (int)ElementCount);

    // 'c' tracks how many bytes have been consumed toward the current UUID string.
    // Starts at 16 to trigger generation on the very first iteration.
    int    c = 16, counter = 0;
    SIZE_T Index = 0;
    char*  UUID = NULL;

    for (int i = 0; i < ShellcodeSize; i++) {

        // Every 16 bytes consumed, generate one UUID string.
        if (c == 16) {
            counter++;

            // Consume bytes [i .. i+15] to form one UUID string.
            UUID = GenerateUUid(
                pShellcode[i],      pShellcode[i + 1],  pShellcode[i + 2],  pShellcode[i + 3],
                pShellcode[i + 4],  pShellcode[i + 5],  pShellcode[i + 6],  pShellcode[i + 7],
                pShellcode[i + 8],  pShellcode[i + 9],  pShellcode[i + 10], pShellcode[i + 11],
                pShellcode[i + 12], pShellcode[i + 13], pShellcode[i + 14], pShellcode[i + 15]
            );
            if (UUID == NULL) {
                // Free already allocated strings before returning.
                if (UuidArray != NULL) {
                    for (SIZE_T j = 0; j < Index; j++)
                        HeapFree(GetProcessHeap(), 0, UuidArray[j]);
                    HeapFree(GetProcessHeap(), 0, UuidArray);
                }
                return FALSE;
            }

            // Save the string pointer only if the caller requested the array.
            if (UuidArray != NULL)
                UuidArray[Index++] = UUID;

            if (i == ShellcodeSize - 16) {
                // Last string — no trailing comma.
                printf("\"%s\"", UUID);

                // If the array is not being saved, free the string now.
                if (UuidArray == NULL)
                    HeapFree(GetProcessHeap(), 0, UUID);

                break;
            } else {
                printf("\"%s\", ", UUID);

                // If the array is not being saved, free the string now.
                if (UuidArray == NULL)
                    HeapFree(GetProcessHeap(), 0, UUID);
            }

            c = 1;

            // Insert a newline every 3 strings to keep output readable.
            if (counter % 3 == 0)
                printf("\n\t");

        } else {
            c++;
        }
    }

    printf("\n};\n\n");

    // Save outputs only if the caller supplied non-NULL pointers.
    if (pUuidArray != NULL)
        *pUuidArray = UuidArray;

    if (pNmbrOfElements != NULL)
        *pNmbrOfElements = ElementCount;

    return TRUE;
}
```

**Example**

```c
unsigned char pShellcode[] = {
    0xFC, 0x48, 0x83, 0xE4, 0xF0, 0xE5, 0xE2, 0x50,
    0x41, 0x51, 0x52, 0x53, 0x56, 0x57, 0x55, 0x53
};  // Must be a multiple of 16 bytes.

CHAR**  pUuidArray     = NULL;
SIZE_T  NmbrOfElements = 0;

// Pass non-NULL pointers to receive the heap-allocated string array.
if (GenerateUuidOutput(pShellcode, sizeof(pShellcode), &pUuidArray, &NmbrOfElements)) {
    for (SIZE_T i = 0; i < NmbrOfElements; i++)
        HeapFree(GetProcessHeap(), 0, pUuidArray[i]);  // Free each string...
    HeapFree(GetProcessHeap(), 0, pUuidArray);           // ...then the array.
}
```
